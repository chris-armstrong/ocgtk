Module Gir_gen_lib.Override_parser

S-expression parser for GIR override files.

Reads override files in the custom sexp format and produces Override_types.library_overrides values.

type parse_error =
  1. | Invalid_format of {
    1. location : string;
    2. message : string;
    }
  2. | Unknown_entity_kind of string
  3. | Unknown_component_kind of {
    1. entity_name : string;
    2. kind : string;
    3. valid_kinds : string list;
    }
  4. | Duplicate_entity of {
    1. kind : string;
    2. name : string;
    }
  5. | Duplicate_component of {
    1. entity : string;
    2. component_kind : string;
    3. name : string;
    }
  6. | Invalid_version of {
    1. name : string;
    2. version : string;
    3. reason : string;
    }

Parse errors with context.
