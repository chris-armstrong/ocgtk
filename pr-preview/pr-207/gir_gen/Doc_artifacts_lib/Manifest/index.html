Module Doc_artifacts_lib.Manifest

The JSON file stored next to each artifact.

type t = {
  1. key : Artifact_key.t;
  2. branch : string;
  3. date : string;
    (*

    ISO 8601, UTC.

    *)
  4. odoc_version : string;
  5. dune_version : string;
  6. byte_size : int;
    (*

    Of the gzipped tar.

    *)
  7. html_file_count : int;
  8. note : string;
  9. odoc_warning_count : int;
}
val to_yojson : t -> Yojson.Safe.t

One object, fields in the order above with sha and dirty after key.

to_yojson, pretty-printed, ending with a newline.
