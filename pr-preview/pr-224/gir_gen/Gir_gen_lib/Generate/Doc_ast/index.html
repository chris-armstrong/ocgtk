type fallback =
  1. | Inline_code_unbalanced of string
    (*

    v1 inline code would break its span — rendered as escaped plain prose.

    *)
  2. | Code_block_verbatim of string
    (*

    v1 code block would break its block delimiter — rendered with the verbatim form instead.

    *)
  3. | Code_block_stripped of string
    (*

    v1 code block contains } and the verbatim fallback would break too — the block is dropped.

    *)
  4. | Sym_ref_degraded of string
    (*

    Fragment / legacy sigil rendered as [endpoint] code, awaiting the §7 resolver leg.

    *)
  5. | Page_ref_degraded of string
    (*

    Relative gi-docgen .html link rendered as bare text, awaiting the §6.3 upstream-URL leg.

    *)
  6. | Link_degraded of string
    (*

    Markdown link to a non-https target — v1 Link is https-only — kept as bare text.

    *)
  7. | Admonition_stripped of string
    (*

    ::: type admonition stripped to its plain content.

    *)
  8. | Table_stripped
    (*

    Markdown table dropped (cell text is not prose).

    *)
  9. | Picture_stripped of string option
    (*

    <picture>/<source> wrappers stripped; the <img alt> text (when present) is kept as plain prose.

    *)
  10. | Image_stripped of string
    (*

    Markdown image or bare <img>: the alt text is kept as plain prose.

    *)
  11. | Quote_stripped
    (*

    > blockquote marker stripped, content kept as plain prose.

    *)
  12. | Ownership_stripped of string
    (*

    A sentence telling C callers how to free a value, dropped; the payload names the matching pattern.

    *)

One lossy or degraded translation, with enough payload to classify and count it for the warnings report (GIR plan Phase 5).
