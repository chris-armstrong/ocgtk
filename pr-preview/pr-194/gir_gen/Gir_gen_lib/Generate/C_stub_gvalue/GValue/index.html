type gvalue_type_category =
  1. | Enum
    (*

    GEnum types

    *)
  2. | Bitfield
    (*

    GFlags/bitfield types

    *)
  3. | Boolean
    (*

    gboolean

    *)
  4. | Integer of {
    1. c_type_name : string;
    2. getter : string;
      (*

      g_value_get_* function name

      *)
    }
    (*

    Integer types (int, uint, long, etc.)

    *)
  5. | Float of string
    (*

    Float/double types with getter name

    *)
  6. | String
    (*

    String types

    *)
  7. | Boxed of {
    1. c_type : string;
    2. is_pointer : bool;
      (*

      true if passed by pointer

      *)
    }
    (*

    Boxed types (records/structs)

    *)
  8. | Object of string
    (*

    Object types with C type name

    *)
  9. | GVariant
    (*

    GLib.Variant — uses g_value_get/set_variant

    *)
  10. | Pointer
    (*

    Generic pointer types

    *)
  11. | Unsupported of string
    (*

    Fallback for unsupported types

    *)
